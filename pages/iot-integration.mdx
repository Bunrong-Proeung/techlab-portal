# 📡 Arduino & Embedded Systems: Lab 01

## ប្រធានបទ៖ ការគ្រប់គ្រង Digital I/O និងបច្ចេកទេស Non-Blocking Delay (`millis()`)

### ១. វត្ថុបំណងនៃការពិសោធន៍ (Objectives)
* យល់ដឹងច្បាស់ពីរចនាសម្ព័ន្ធ Hardware នៃ GPIO Pins លើ Microcontroller (ATmega328P / ESP32)
* ស្វែងយល់ពីតួនាទីរបស់ Internal Pull-Up Resistor (`INPUT_PULLUP`)
* យល់ដឹងពីផលប៉ះពាល់នៃការប្រើ `delay()` (CPU Blocking) និងរៀនប្រើ `millis()` សម្រាប់ Multitasking

### ៣. គ្រោងសៀគ្វីពិសោធន៍ (Hardware Wiring)

* **LED 1 (Blinking LED):** 
  * ជើង Anode (+) $\rightarrow$ រេស៊ីស្តង់ 220Ω $\rightarrow$ Pin `D13`
  * ជើង Cathode (-) $\rightarrow$ ជើងដី `GND`
* **LED 2 (Indicator LED):**
  * ជើង Anode (+) $\rightarrow$ រេស៊ីស្តង់ 220Ω $\rightarrow$ Pin `D12`
  * ជើង Cathode (-) $\rightarrow$ ជើងដី `GND`
* **Push Button:**
  * ជើងទី ១ $\rightarrow$ ជើងដី `GND`
  * ជើងទី ២ $\rightarrow$ Pin `D2` *(កំណត់ជា `INPUT_PULLUP` ក្នុងកូដ មិនបាច់ប្រើរេស៊ីស្តង់ខាងក្រៅឡើយ)*

### ៤. កូដគំរូកម្មវិធី (C/C++ Source Code)

កូដនេះបង្ហាញពីការដំណើរការកិច្ចការ ២ ក្នុងពេលតែមួយ (Multitasking) ដោយ LED 1 ភ្លឹបភ្លែតជាប្រចាំ ខណៈដែល Button អាចបញ្ជា LED 2 បានភ្លាមៗដោយគ្មានការរអាក់រអួល៖

```cpp
// កំណត់ឈ្មោះជើង Pin
const int LED_BLINK_PIN = 13;
const int LED_STATE_PIN = 12;
const int BUTTON_PIN    = 2;

// អថេរសម្រាប់គ្រប់គ្រងពេលវេលា (Non-blocking timing)
unsigned long previousMillis = 0;
const long interval = 500; // រយៈពេលភ្លឹបភ្លែត 500ms

int ledBlinkState = LOW;

void setup() {
  // កំណត់ទិសដៅជើង I/O
  pinMode(LED_BLINK_PIN, OUTPUT);
  pinMode(LED_STATE_PIN, OUTPUT);
  
  // ប្រើប្រាស់ Internal Pull-Up Resistor (Logic ធម្មតា = HIGH, ពេលចុច = LOW)
  pinMode(BUTTON_PIN, INPUT_PULLUP);

  Serial.begin(9600);
  Serial.println("System Ready: Non-blocking Lab Initialized.");
}

void loop() {
  unsigned long currentMillis = millis();

  // --- កិច្ចការទី ១៖ បញ្ជា LED 1 ឱ្យភ្លឹបភ្លែតដោយមិនប្រើ delay() ---
  if (currentMillis - previousMillis >= interval) {
    previousMillis = currentMillis; // កត់ត្រាពេលវេលាថ្មី
    
    // ប្ដូរស្ថានភាព LED (Toggle)
    ledBlinkState = (ledBlinkState == LOW) ? HIGH : LOW;
    digitalWrite(LED_BLINK_PIN, ledBlinkState);
  }

  // --- កិច្ចការទី ២៖ អានស្ថានភាពប៊ូតុងចុចភ្លាមៗ (Real-time Response) ---
  int buttonState = digitalRead(BUTTON_PIN);

  if (buttonState == LOW) {
    // នៅពេលចុចប៊ូតុង ជើង Pin នឹងភ្ជាប់ទៅ GND (Active-LOW)
    digitalWrite(LED_STATE_PIN, HIGH);
  } else {
    digitalWrite(LED_STATE_PIN, LOW);
  }
}
